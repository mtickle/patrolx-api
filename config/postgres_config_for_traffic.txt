CREATE OR REPLACE PROCEDURE public.addtraffic(
	IN p_belts text,
	IN p_location text,
	IN p_race text,
	IN p_arresttype text,
	IN p_charge text,
	IN p_subagency text,
	IN p_dateofstop text,
	IN p_color text,
	IN p_vehicletype text,
	IN p_accident text,
	IN p_state text,
	IN p_violationtype text,
	IN p_latitude text,
	IN p_driverstate text,
	IN p_model text,
	IN p_personalinjury text,
	IN p_article text,
	IN p_description text,
	IN p_hazmat text,
	IN p_fatal text,
	IN p_year text,
	IN p_propertydamage text,
	IN p_agency text,
	IN p_gender text,
	IN p_drivercity text,
	IN p_longitude text,
	IN p_alcohol text,
	IN p_timeofstop text,
	IN p_commercialvehicle text,
	IN p_make text,
	IN p_workzone text,
	IN p_dlstate text,
	IN p_contributedtoaccident text,
	IN p_commerciallicense text)
LANGUAGE 'plpgsql'
AS $BODY$
BEGIN

INSERT INTO traffic (
        belts 
, location 
, race 
, arresttype 
, charge 
, subagency 
, dateofstop 
, color 
, vehicletype 
, accident 
, state 
, violationtype 
, latitude 
, driverstate 
, model 
, personalinjury 
, article 
, description 
, hazmat 
, fatal 
, year 
, propertydamage 
, agency 
, gender 
, drivercity 
, longitude 
, alcohol 
, timeofstop 
, commercialvehicle 
, make 
, workzone 
, dlstate 
, contributedtoaccident 
, commerciallicense 
    ) VALUES (
  p_belts
,  p_location
,  p_race
,  p_arresttype
,  p_charge
,  p_subagency
,  p_dateofstop
,  p_color
,  p_vehicletype
,  p_accident
,  p_state
,  p_violationtype
,  p_latitude
,  p_driverstate
,  p_model
,  p_personalinjury
,  p_article
,  p_description
,  p_hazmat
,  p_fatal
,  p_year
,  p_propertydamage
,  p_agency
,  p_gender
,  p_drivercity
,  p_longitude
,  p_alcohol
,  p_timeofstop
,  p_commercialvehicle
,  p_make
,  p_workzone
,  p_dlstate
,  p_contributedtoaccident
,  p_commerciallicense
    )
ON CONFLICT (dateofstop, location) DO NOTHING;

END;
$BODY$;
