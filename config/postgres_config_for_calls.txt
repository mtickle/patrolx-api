-- Table: public.calls
-- DROP TABLE IF EXISTS public.calls;

CREATE TABLE IF NOT EXISTS public.calls
(
    call_id uuid DEFAULT gen_random_uuid(),
    call_agency character varying COLLATE pg_catalog."default",
    call_latitude numeric,
    call_longitude numeric,
    call_type character varying COLLATE pg_catalog."default",
    call_address character varying COLLATE pg_catalog."default",
    call_date date,
    call_time time without time zone,
    CONSTRAINT const_calls UNIQUE (call_agency, call_type, call_address, call_date, call_time)
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS public.calls
    OWNER to pi;

-- FUNCTION: public.getcallcountsbyagency(integer)

-- DROP FUNCTION IF EXISTS public.getcallcountsbyagency(integer);

CREATE OR REPLACE FUNCTION public.getcallcountsbyagency(
	recordlimit integer)
    RETURNS TABLE(item character varying, itemcount bigint) 
    LANGUAGE 'plpgsql'
    COST 100
    VOLATILE PARALLEL UNSAFE
    ROWS 1000

AS $BODY$
BEGIN
    RETURN QUERY
select
	agency as item
	, count(agency) as itemcount
from 
	calls
group by 
	agency
order by
	itemcount desc
limit 10;
END;
$BODY$;

ALTER FUNCTION public.getcallcountsbyagency(integer)
    OWNER TO pi;


select agency, incident, calldate, count(*)
from calls
group by agency, incident, calldate
HAVING count(*) > 1

    calls a
        USING calls b
WHERE
    a.agency = b.agency
    AND a.incident = b.incident
